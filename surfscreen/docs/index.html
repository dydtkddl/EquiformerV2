🚀 빠른 시작

# 1. 설치 pip install surfscreen # 2. 표면 생성 surfscreen surface create Cu --miller 111 --supercell 3x3x1 --layers 4 # 3. 분자 준비 surfscreen molecule from-pubchem acetone -o acetone.xyz # 4. 스크리닝 실행 surfscreen screen run -s Cu111.extxyz -m acetone.xyz --engine mace -o results/ # 5. 리포트 생성 surfscreen screen report results/ -o report.html
